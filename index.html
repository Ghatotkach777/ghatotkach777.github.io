Practical 1.2 - Modules

Creating and using a local module in Node.js

calc.js
exports.add = function(x, y) {
    return x + y;
};

exports.sub = function(x, y) {
    return x - y;
};

exports.mult = function(x, y) {
    return x * y;
};

exports.div = function(x, y) {
    return x / y;
};
index.js
var calculator = require('./calc');

var y = 10;

console.log("Addition of 50 and 10 is " + calculator.add(x, y));

console.log("Division of 50 and 10 is " + calculator.div(x, y));

Practical 1.3 - Creating an HTTP Web Server

Creating a basic HTTP server using Node.js

var http = require('http');

http.createServer(function(req, res) {

    res.writeHead(200, {
        'Content-Type': 'text/html'
    });

    res.write('Welcome to this page !');

    res.end();

}).listen(3000);

Practical 1.5 - Routing in Node.js

Handling different URLs using routing

var http = require('http');

http.createServer(function(req, res) {

    if (req.url == "/") {

        res.writeHead(200, {
            'Content-Type': 'text/html'
        });

        res.end("Home Page");

    }

    else if (req.url == "/about") {

        res.writeHead(200, {
            'Content-Type': 'text/html'
        });

        res.end("About Page");

    }

    else {

        res.writeHead(404, {
            'Content-Type': 'text/html'
        });

        res.end("Page Not Found");
    }

}).listen(3000);

console.log("Server Running...");
