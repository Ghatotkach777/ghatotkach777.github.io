Practical 3

Area and Circumference of Circle

Area = πr² • Circumference = 2πr
SET SERVEROUTPUT ON;

DECLARE
    r NUMBER := 6;
    pi NUMBER := 3.14;
    area NUMBER;
    circumference NUMBER;
BEGIN
    area := pi * r * r;
    circumference := 2 * pi * r;

    DBMS_OUTPUT.PUT_LINE('Area of circle = ' || area);
    DBMS_OUTPUT.PUT_LINE('Circumference of circle = ' || circumference);
END;

Practical Set 2 — Conditional PL/SQL

Practical 1

Check Even or Odd

MOD(num, 2) = 0 → even
SET SERVEROUTPUT ON;

DECLARE
    num INTEGER := &num;
BEGIN
    IF MOD(num, 2) = 0 THEN
        DBMS_OUTPUT.PUT_LINE('Number is even: ' || num);
    ELSE
        DBMS_OUTPUT.PUT_LINE('Number is odd: ' || num);
    END IF;
END;
/
Practical 2

Discount Eligibility by Age

Age ≤ 7 OR age ≥ 60 → 30% discount
SET SERVEROUTPUT ON;

DECLARE
    age NUMBER := &age;
BEGIN
    IF age <= 7 OR age >= 60 THEN
        DBMS_OUTPUT.PUT_LINE('You will have 30% discount');
    ELSE
        DBMS_OUTPUT.PUT_LINE('You are not eligible for discount');
    END IF;
END;

Practical 3

Check for a Substring

Uses LOWER() and LIKE
SET SERVEROUTPUT ON;

DECLARE
    v_main_string VARCHAR2(200) := 'Diagnosis of thyroid disorder of Dolly Shah';
    v_substring   VARCHAR2(50) := 'health';
BEGIN
    IF LOWER(v_main_string) LIKE '%' || LOWER(v_substring) || '%' THEN
        DBMS_OUTPUT.PUT_LINE('Substring found');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Substring not found');
    END IF;
END;
/
Change v_substring to the word you want to search for.
