Practical — Prog21 — Sum of Digits

class Prog21
{
    public static void main(String args[])
    {
        int n,i,sum=0,rem=0;
        n=87654;

        for(i=1; i<=5; i++)
        {
            rem=n%10;
            sum=sum+rem;
            n=n/10;
        }

        System.out.println("Sum of 5 digit number is "+sum);
    }
}

Practical — Prog16 — Factorial Using Do-While

class Prog16
{
    public static void main(String args[])
    {
        double n,i,fact;
        fact=1;
        n=20;
        i=1;

        do
        {
            fact=fact*i;
            i++;
        }
        while(i<=n);

        System.out.println("fact is "+n+" is "+fact);
    }
}

Practical — Prog24 — Number Pattern

class Prog24
{
    public static void printPattern(int n)
    {
        int i,j;

        for(i=1; i<=n; i++)
        {
            for(j=1; j<=n-i; j++)
            {
                System.out.print(" ");
            }

            for(j=1; j<=i; j++)
            {
                System.out.print(i+" ");
            }

            System.out.println();
        }
    }

    public static void main(String args[])
    {
        int n=6;
        printPattern(6);
    }
}
